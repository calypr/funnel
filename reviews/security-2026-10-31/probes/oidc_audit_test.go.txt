type auditTransport func(*http.Request)(*http.Response,error)
func(f auditTransport)RoundTrip(r *http.Request)(*http.Response,error){return f(r)}
func auditResponse(r *http.Request,body string)*http.Response{return &http.Response{StatusCode:200,Header:http.Header{"Content-Type":[]string{"application/json"}},Body:io.NopCloser(strings.NewReader(body)),Request:r}}
func TestAuditOIDCStartup(t *testing.T){
 previous:=http.DefaultTransport;defer func(){http.DefaultTransport=previous}()
 http.DefaultTransport=auditTransport(func(r *http.Request)(*http.Response,error){if strings.Contains(r.URL.Path,"jwks"){return auditResponse(r,`{"keys":[]}`),nil};return auditResponse(r,`{"issuer":"https://issuer.invalid","jwks_uri":"https://issuer.invalid/jwks","authorization_endpoint":"https://issuer.invalid/auth","token_endpoint":"https://issuer.invalid/token"}`),nil})
 c:=&OidcConfig{local:&config.OidcAuth{ServiceConfigURL:"https://issuer.invalid/config",ClientId:"funnel",ClientSecret:"synthetic",RedirectURL:"https://funnel.invalid/login"}}
 defer func(){if c.jwks!=nil{_ = c.jwks.Shutdown(context.Background())};if p:=recover();p!=nil{t.Logf("configured OIDC initialization panicked: %v",p)}else{t.Fatal("expected suspected startup failure did not occur")}}()
 c.initConfig()
}
func TestAuditOIDCLoginBinding(t *testing.T){
 previous:=http.DefaultTransport;defer func(){http.DefaultTransport=previous}()
 http.DefaultTransport=auditTransport(func(r *http.Request)(*http.Response,error){return auditResponse(r,`{"access_token":"synthetic-other-session-token","token_type":"Bearer","expires_in":3600}`),nil})
 c:=&OidcConfig{local:&config.OidcAuth{ServiceConfigURL:"https://issuer.invalid/config",ClientId:"funnel",ClientSecret:"synthetic"},oauth2:&oauth2.Config{ClientID:"funnel",ClientSecret:"synthetic",Endpoint:oauth2.Endpoint{AuthURL:"https://issuer.invalid/auth",TokenURL:"https://issuer.invalid/token"}}}
 start:=httptest.NewRequest("GET","https://funnel.invalid/login",nil);start.Header.Set("User-Agent","synthetic-browser")
 first:=httptest.NewRecorder();c.RedirectToLogin(first,start);location,_:=url.Parse(first.Header().Get("Location"))
 fresh:=httptest.NewRequest("GET","https://funnel.invalid/login?state="+location.Query().Get("state")+"&code=synthetic-code",nil);fresh.Header.Set("User-Agent","synthetic-browser")
 out:=httptest.NewRecorder();c.HandleAuthCode(out,fresh)
 t.Logf("fresh session cookies=%d callback status=%d sets-session-cookie=%t",len(fresh.Cookies()),out.Code,strings.Contains(out.Header().Get("Set-Cookie"),"jwt=synthetic-other-session-token"))
 if out.Code!=307||!strings.Contains(out.Header().Get("Set-Cookie"),"jwt=synthetic-other-session-token"){t.Fatal("suspected cross-session callback acceptance not observed")}
 different:=httptest.NewRequest("GET",fresh.URL.String(),nil);different.Header.Set("User-Agent","different-browser");deny:=httptest.NewRecorder();c.HandleAuthCode(deny,different);t.Logf("different user agent callback status=%d",deny.Code)
}

 previous:=http.DefaultTransport;defer func(){http.DefaultTransport=previous}();http.DefaultTransport=auditTransport(func(r *http.Request)(*http.Response,error){return auditResponse(r,string(jwks)),nil})

 cases:=[]struct{name,aud string;exp any;required string}{{"wrong audience","another-client",time.Now().Add(time.Hour).Unix(),""},{"missing expiry","funnel",nil,"funnel"},{"expired control","funnel",time.Now().Add(-time.Hour).Unix(),"funnel"},{"wrong audience with explicit policy","another-client",time.Now().Add(time.Hour).Unix(),"funnel"}}
 for _,tc:=range cases{claims:=map[string]any{"iss":"https://issuer.invalid","sub":"synthetic-user","aud":tc.aud};if tc.exp!=nil{claims["exp"]=tc.exp};c.local.RequireAudience=tc.required;got:=c.ParseJwtSubject(signOIDCTestToken(t,key,claims));t.Logf("%s accepted=%t",tc.name,got!="")}
}
