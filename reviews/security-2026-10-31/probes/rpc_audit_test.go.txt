func(w *auditWriter)Close(){}

 conn,err:=rpc.Dial(ctx,c,grpc.WithContextDialer(func(context.Context,string)(net.Conn,error){return lis.Dial()}));if err!=nil{t.Fatal(err)};defer conn.Close()
 _,err=events.NewEventServiceClient(conn).WriteEvent(ctx,events.NewMetadata("synthetic-task",0,map[string]string{"test":"value"}));if err!=nil{t.Fatal(err)};t.Log(w.observation)
}
