package server
import("context";"net/http/httptest";"strings";"testing";"github.com/calypr/funnel/tes";"github.com/grpc-ecosystem/grpc-gateway/v2/runtime";"google.golang.org/grpc";"google.golang.org/grpc/codes";"google.golang.org/grpc/status")
type auditRejectingClient struct{tes.TaskServiceClient;seen int}
func(c *auditRejectingClient)CreateTask(ctx context.Context,task *tes.Task,opts ...grpc.CallOption)(*tes.CreateTaskResponse,error){c.seen=len(task.Description);return nil,status.Error(codes.Unauthenticated,"synthetic auth rejection")}
func TestAuditGatewayBuffersBeforeAuth(t *testing.T){c:=&auditRejectingClient{};mux:=runtime.NewServeMux(runtime.WithMarshalerOption(runtime.MIMEWildcard,NewMarshaler()));if err:=tes.RegisterTaskServiceHandlerClient(context.Background(),mux,c);err!=nil{t.Fatal(err)};body:=`{"description":"`+strings.Repeat("x",5<<20)+`"}`;req:=httptest.NewRequest("POST","/v1/tasks",strings.NewReader(body));req.Header.Set("Content-Type","application/json");out:=httptest.NewRecorder();mux.ServeHTTP(out,req);t.Logf("authorization_header=%q fully_decoded_description_bytes=%d response_status=%d",req.Header.Get("Authorization"),c.seen,out.Code);if c.seen!=5<<20||out.Code!=401{t.Fatal("unexpected gateway behavior")}}
