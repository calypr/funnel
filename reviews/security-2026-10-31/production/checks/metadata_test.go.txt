package tes

import "testing"

func TestAuditServerMetadataProvenance(t *testing.T) {
	task := &Task{Executors: []*Executor{{Image: "synthetic", Command: []string{"true"}}}, Logs: []*TaskLog{{Metadata: map[string]string{"awsbatch_id": "synthetic-client-supplied-id", "slurm_id": "synthetic-client-supplied-id"}}}}
	if err := InitTask(task, true); err != nil {
		t.Fatal(err)
	}
	t.Logf("client_logs_retained=%t client_backend_id_retained=%t", len(task.Logs) > 0, task.Logs[0].Metadata["awsbatch_id"] == "synthetic-client-supplied-id")
}
