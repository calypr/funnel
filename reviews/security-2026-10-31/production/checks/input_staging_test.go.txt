package worker

import (
	"context"
	"github.com/calypr/funnel/config"
	"github.com/calypr/funnel/storage"
	"github.com/calypr/funnel/tes"
	"os"
	"path/filepath"
	"testing"
)

func TestAuditInputStagingIndependence(t *testing.T) {
	base := t.TempDir()
	shared := filepath.Join(base, "shared")
	if err := os.MkdirAll(shared, 0700); err != nil {
		t.Fatal(err)
	}
	original := filepath.Join(shared, "fixture.txt")
	if err := os.WriteFile(original, []byte("synthetic input fixture"), 0600); err != nil {
		t.Fatal(err)
	}
	mapper := NewFileMapper(filepath.Join(base, "workspace"))
	task := &tes.Task{Executors: []*tes.Executor{{Image: "synthetic", Command: []string{"true"}}}, Inputs: []*tes.Input{{Url: original, Path: "/inputs/fixture.txt", Type: tes.File}}}
	if err := mapper.MapTask(task); err != nil {
		t.Fatal(err)
	}
	backend, err := storage.NewLocal(&config.LocalStorage{AllowedDirs: []string{shared}})
	if err != nil {
		t.Fatal(err)
	}
	if _, err = backend.Get(context.Background(), original, mapper.Inputs[0].Path); err != nil {
		t.Fatal(err)
	}
	src, err := os.Stat(original)
	if err != nil {
		t.Fatal(err)
	}
	dst, err := os.Stat(mapper.Inputs[0].Path)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("source_and_staged_input_same_inode=%t executor_input_mount_readonly=%t", os.SameFile(src, dst), mapper.Volumes[0].Readonly)
	if !os.SameFile(src, dst) || mapper.Volumes[0].Readonly {
		t.Fatal("recorded staging behavior not reproduced")
	}
}
