package worker

import (
	"context"
	"github.com/calypr/funnel/tes"
	"os"
	"path/filepath"
	"testing"
)

func TestAuditOutputLinkClassification(t *testing.T) {
	root := t.TempDir()
	workspace := filepath.Join(root, "workspace")
	if err := os.Mkdir(workspace, 0700); err != nil {
		t.Fatal(err)
	}
	target := filepath.Join(root, "fixture.txt")
	if err := os.WriteFile(target, []byte("fixture"), 0600); err != nil {
		t.Fatal(err)
	}
	link := filepath.Join(workspace, "result.txt")
	if err := os.Symlink(target, link); err != nil {
		t.Fatal(err)
	}
	flat, err := FlattenOutputs(context.Background(), []*tes.Output{{Path: link, Url: "synthetic://record-only", Type: tes.File}}, nil, nil)
	if err != nil {
		t.Fatal(err)
	}
	st, err := os.Lstat(flat[0].Path)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("outside_workspace_symlink_accepted_for_upload=%t", st.Mode()&os.ModeSymlink != 0)
	if st.Mode()&os.ModeSymlink == 0 {
		t.Fatal("recorded classification behavior not reproduced")
	}
}
