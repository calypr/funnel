package server

import (
	"context"
	"github.com/calypr/funnel/config"
	"github.com/calypr/funnel/logger"
	apisrv "github.com/calypr/funnel/server"
	"google.golang.org/grpc"
	"net"
	"path/filepath"
	"testing"
)

func TestAuditDefaultExposurePolicy(t *testing.T) {
	c := config.DefaultConfig()
	c.Compute = "noop"
	c.BoltDB.Path = filepath.Join(t.TempDir(), "audit.db")
	c.EventWriters = nil
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	l := logger.NewLogger("audit", logger.DefaultConfig())
	l.Discard()
	assembled, err := NewServer(ctx, c, l)
	if err != nil {
		t.Fatal(err)
	}
	defer assembled.Server.Tasks.(*apisrv.TaskService).Read.Close()
	rpcAddr, err := net.ResolveTCPAddr("tcp", assembled.RPCAddress)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("advertised_host=%s rpc_bind=%s wildcard_ip=%t basic_credentials=%d oidc_configured=%t", c.Server.HostName, assembled.RPCAddress, len(rpcAddr.IP) == 0, len(c.Server.BasicAuth), c.Server.OidcAuth != nil && c.Server.OidcAuth.ServiceConfigURL != "")
	auth := apisrv.NewAuthentication(assembled.BasicAuth, assembled.OidcAuth, assembled.TaskAccess)
	_, err = auth.Interceptor(context.Background(), nil, &grpc.UnaryServerInfo{FullMethod: "/audit/policy"}, func(ctx context.Context, _ any) (any, error) {
		u := apisrv.GetUser(ctx)
		t.Logf("no_credentials_public=%t all_tasks=%t other_owner_accessible=%t", u.IsPublic, u.CanSeeAllTasks(), u.IsAccessible("synthetic-other-owner"))
		return nil, nil
	})
	if err != nil {
		t.Fatal(err)
	}
}
