//go:build unix

package storage

import (
	"context"
	"errors"
	"github.com/calypr/funnel/config"
	"golang.org/x/sys/unix"
	"net/http"
	"path/filepath"
	"testing"
	"time"
)

type reviewNoNetworkTransport struct{}

func (reviewNoNetworkTransport) RoundTrip(*http.Request) (*http.Response, error) {
	return nil, errors.New("synthetic transport: network disabled")
}
func TestReviewUploadSourceCancellation(t *testing.T) {
	fixture := filepath.Join(t.TempDir(), "fixture.fifo")
	if err := unix.Mkfifo(fixture, 0600); err != nil {
		t.Fatal(err)
	}
	backend, err := NewHTTP(config.DefaultConfig().HTTPStorage)
	if err != nil {
		t.Fatal(err)
	}
	backend.client.Transport = reviewNoNetworkTransport{}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	done := make(chan error, 1)
	go func() { _, err := backend.Put(ctx, "https://example.invalid/not-sent", fixture); done <- err }()
	time.Sleep(100 * time.Millisecond)
	cancel()
	pending := false
	select {
	case <-done:
	case <-time.After(100 * time.Millisecond):
		pending = true
	}
	t.Logf("canceled_upload_pending_on_nonregular_source=%t", pending)
	if !pending {
		t.Fatal("recorded source-opening behavior not reproduced")
	}
	// Release the temporary FIFO so the check leaves no blocked operation.
	fd, err := unix.Open(fixture, unix.O_WRONLY|unix.O_NONBLOCK, 0)
	if err != nil {
		t.Fatal(err)
	}
	if err = unix.Close(fd); err != nil {
		t.Fatal(err)
	}
	select {
	case err := <-done:
		t.Logf("operation_finished_after_fixture_release=%t returned_error=%t", true, err != nil)
	case <-time.After(time.Second):
		t.Fatal("temporary fixture operation did not finish")
	}
}
