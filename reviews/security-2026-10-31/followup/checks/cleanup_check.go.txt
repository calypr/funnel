package main
import (
 "context"
 "fmt"
 "github.com/calypr/funnel/compute/kubernetes/resources"
 "github.com/calypr/funnel/config"
 "github.com/calypr/funnel/logger"
 batch "k8s.io/api/batch/v1"
 core "k8s.io/api/core/v1"
 errors "k8s.io/apimachinery/pkg/api/errors"
 meta "k8s.io/apimachinery/pkg/apis/meta/v1"
 "k8s.io/client-go/kubernetes/fake"
)
func main(){
 ctx:=context.Background(); log:=logger.NewLogger("review",logger.DefaultConfig()); conf:=config.DefaultConfig(); conf.Kubernetes.JobsNamespace="synthetic"
 client:=fake.NewSimpleClientset(&batch.Job{ObjectMeta:meta.ObjectMeta{Name:"task-a-operator-backup",Namespace:"synthetic",Labels:map[string]string{"app":"operator"}}},&core.ServiceAccount{ObjectMeta:meta.ObjectMeta{Name:"external-operator-sa",Namespace:"synthetic",Labels:map[string]string{"owner":"operator"}}})
 if err:=resources.DeleteExecutorJobs(ctx,conf,"task-a",client,log);err!=nil{panic(err)}
 _,err:=client.BatchV1().Jobs("synthetic").Get(ctx,"task-a-operator-backup",meta.GetOptions{});fmt.Printf("unrelated_prefix_job_removed=%t\n",errors.IsNotFound(err))
 if err:=resources.DeleteServiceAccount(ctx,"task-a","synthetic",client,log,&resources.DeleteServiceAccountOptions{ServiceAccountName:"external-operator-sa",SharedSA:true});err!=nil{panic(err)}
 _,err=client.CoreV1().ServiceAccounts("synthetic").Get(ctx,"external-operator-sa",meta.GetOptions{});fmt.Printf("external_idle_shared_sa_removed=%t\n",errors.IsNotFound(err))
}
