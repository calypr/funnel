package server

import (
	"context"
	"github.com/calypr/funnel/config"
	"github.com/calypr/funnel/logger"
	plugin "github.com/calypr/funnel/plugins/proto"
	"github.com/calypr/funnel/tes"
	"google.golang.org/grpc/metadata"
	"testing"
)

type reviewCancelPolicy struct {
	called bool
	task   *tes.Task
	action plugin.Type
}

func (p *reviewCancelPolicy) PluginAction(_ map[string]string, _ map[string]*plugin.StringList, _ *config.Config, task *tes.Task, action plugin.Type) (*plugin.JobResponse, error) {
	p.called = true
	p.task = task
	p.action = action
	return &plugin.JobResponse{Code: 200, Config: config.DefaultConfig()}, nil
}

type reviewTerminalReader struct{ tes.ReadOnlyServer }

func (reviewTerminalReader) GetTask(context.Context, *tes.GetTaskRequest) (*tes.Task, error) {
	return &tes.Task{Id: "synthetic-object", State: tes.Complete}, nil
}
func TestReviewCancelPolicyTarget(t *testing.T) {
	c := config.DefaultConfig()
	c.Plugins = &config.Plugins{Path: "synthetic"}
	policy := &reviewCancelPolicy{}
	log := logger.NewLogger("review", logger.DefaultConfig())
	log.Discard()
	svc := &TaskService{Config: c, Plugin: policy, Read: reviewTerminalReader{}, Log: log}
	ctx := metadata.NewIncomingContext(context.Background(), metadata.Pairs("x-review", "synthetic"))
	_, err := svc.CancelTask(ctx, &tes.CancelTaskRequest{Id: "synthetic-object"})
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("cancel_policy_called=%t action=%s target_task_missing=%t", policy.called, policy.action.String(), policy.task == nil)
	if !policy.called || policy.action != plugin.Type_CANCEL || policy.task != nil {
		t.Fatal("recorded policy argument omission not reproduced")
	}
}
