//go:build unix

package fsutil

import (
	"os"
	"path/filepath"
	"syscall"
	"testing"
)

func TestReviewDirectoryPreservesCallerPermissions(t *testing.T) {
	previous := syscall.Umask(0077)
	defer syscall.Umask(previous)
	base := t.TempDir()
	if err := EnsureDir(filepath.Join(base, "new-directory")); err != nil {
		t.Fatal(err)
	}
	after := syscall.Umask(0)
	syscall.Umask(after)
	file := filepath.Join(base, "ordinary-file.txt")
	if err := os.WriteFile(file, []byte("fixture"), 0666); err != nil {
		t.Fatal(err)
	}
	info, err := os.Stat(file)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("caller_umask_before=0077 caller_umask_after=%04o subsequent_file_mode=%04o", after, info.Mode().Perm())
	if after != 0 || info.Mode().Perm() != 0666 {
		t.Fatal("recorded permission behavior not reproduced")
	}
}
